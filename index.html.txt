function startHeroAnimation() {
  const title = document.getElementById('heroTitle');
  const text = "I'm Fizza Rabbani";
  title.innerHTML = '';
  let delay = 0.3;
  text.split(' ').forEach(word => {
    const w = document.createElement('span');
    w.className = 'word';
    [...word].forEach(ch => {
      const i = document.createElement('i');
      i.textContent = ch;
      i.style.animationDelay = delay + 's';
      w.appendChild(i);
      delay += 0.05;
    });
    w.innerHTML += '&nbsp;';
    title.appendChild(w);
  });
}

/* ===== ⌨️ TYPING EFFECT ===== */
const roles = ['Graphic Designer 🎨', 'Video Editor 🎬', 'AI Learner 🤖', 'AD CS Student 💻', 'Creative Developer ✨'];
let ri = 0, ci = 0, deleting = false;
function typeLoop() {
  const el = document.getElementById('typed');
  const word = roles[ri];
  el.textContent = word.slice(0, ci);
  if (!deleting && ci < word.length) { ci++; setTimeout(typeLoop, 80); }
  else if (deleting && ci > 0) { ci--; setTimeout(typeLoop, 40); }
  else if (!deleting) { deleting = true; setTimeout(typeLoop, 1600); }
  else { deleting = false; ri = (ri + 1) % roles.length; setTimeout(typeLoop, 300); }
}
typeLoop();

/* ===== ✨ GLOWING PARTICLES ===== */
const canvas = document.getElementById('stars');
const ctx = canvas.getContext('2d');
function resize() { canvas.width = innerWidth; canvas.height = innerHeight; }
resize();
addEventListener('resize', resize);

const COLORS = ['#60a5fa', '#22d3ee', '#a78bfa', '#ffffff'];
const particles = [];
for (let i = 0; i < 70; i++) {
  particles.push({
    x: Math.random() * innerWidth,
    y: Math.random() * innerHeight,
    r: Math.random() * 2 + 0.5,
    s: Math.random() * 0.4 + 0.1,
    c: COLORS[Math.floor(Math.random() * COLORS.length)],
    tw: Math.random() * Math.PI * 2
  });
}
function drawStars() {
  ctx.clearRect(0, 0, canvas.width, canvas.height);
  particles.forEach(p => {
    p.y -= p.s; p.tw += 0.03;
    if (p.y < -10) { p.y = canvas.height + 10; p.x = Math.random() * innerWidth; }
    ctx.globalAlpha = Math.max(0.35 + Math.sin(p.tw) * 0.3, 0.06);
    ctx.fillStyle = p.c;
    ctx.shadowBlur = 6;
    ctx.shadowColor = p.c;
    ctx.beginPath();
    ctx.arc(p.x, p.y, p.r, 0, Math.PI * 2);
    ctx.fill();
  });
  ctx.globalAlpha = 1;
  requestAnimationFrame(drawStars);
}
drawStars();

/* ===== 🎯 REVEAL + COUNTERS ===== */
const observer = new IntersectionObserver(entries => {
  entries.forEach(e => {
    if (!e.isIntersecting) return;
    e.target.classList.add('show');
  });
}, { threshold: 0.15 });
document.querySelectorAll('.reveal, .skill-card').forEach(el => observer.observe(el));

const counterObs = new IntersectionObserver(entries => {
  entries.forEach(e => {
    if (!e.isIntersecting) return;
    const el = e.target, target = +el.dataset.target;
    let cur = 0;
    const step = Math.max(1, Math.round(target / 60));
    const t = setInterval(() => {
      cur += step;
      if (cur >= target) { cur = target; clearInterval(t); }
      el.textContent = cur + (target === 100 ? '' : '+');
    }, 30);
    counterObs.unobserve(el);
  });
}, { threshold: 0.5 });
document.querySelectorAll('.counter').forEach(el => counterObs.observe(el));
